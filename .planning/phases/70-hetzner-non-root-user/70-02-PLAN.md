Run as root on the Hetzner VPS:

```bash
# 1. Register all three PM2 processes under the claude user
sudo -u claude -i bash << 'CLAUDEBLOCK'
  set -e

  # gsd-dashboard (Node.js server)
  pm2 start /home/services/gsddashboard/server/index.js \
    --name gsd-dashboard \
    --cwd /home/services/gsddashboard

  # gsd-healthcheck (shell script loop)
  pm2 start /home/services/gsddashboard/scripts/healthcheck.sh \
    --name gsd-healthcheck \
    --interpreter sh \
    --cwd /home/services/gsddashboard

  # gsd-tunnel (cloudflared via named-tunnel.sh)
  pm2 start /home/services/gsddashboard/scripts/named-tunnel.sh \
    --name gsd-tunnel \
    --interpreter sh \
    --cwd /home/services/gsddashboard

  # Confirm all three are running
  pm2 list

  # Persist the process list
  pm2 save
CLAUDEBLOCK

# 2. Generate and immediately run the systemd startup command.
#    pm2 startup prints a "sudo env PATH=..." line — capture and run it as root.
STARTUP_CMD=$(sudo -u claude -i pm2 startup systemd -u claude --hp /home/claude 2>&1 | grep "sudo env")
echo "Installing systemd unit via: $STARTUP_CMD"
eval "$STARTUP_CMD"

# 3. Enable and start the service
systemctl enable pm2-claude.service
systemctl start pm2-claude.service

# 4. Wait a moment then verify
sleep 3
systemctl status pm2-claude.service | head -20
curl -s -o /dev/null -w "Dashboard HTTP: %{http_code}\n" http://localhost:4820/api/health
```

```bash
# 1. Install root's crontab for claude user.
#    The KidAI cron jobs reference /home/services/KidAI paths — ownership was already
#    transferred to claude in Plan 01, so they run correctly.
if [ -s /tmp/root-crontab.txt ]; then
  crontab -u claude /tmp/root-crontab.txt
  echo "=== CLAUDE CRONTAB ===" && crontab -u claude -l
else
  echo "Root crontab was empty — nothing to migrate"
fi

# 2. Remove root's crontab to avoid duplicate execution
crontab -r 2>/dev/null && echo "Root crontab removed" || echo "No root crontab to remove"

#    The GSD Dashboard server (now running as claude via PM2) will create new tmux

#    Root-owned sessions at /tmp/tmux-0/ are inaccessible to the claude process.
tmux kill-server 2>/dev/null && echo "Root tmux server killed" || echo "No root tmux server running"

# 4. Verify
ls /tmp/tmux-0/ 2>/dev/null && echo "WARNING: /tmp/tmux-0/ still exists" || echo "PASS: root tmux socket gone"
crontab -l 2>/dev/null && echo "WARNING: root crontab still present" || echo "PASS: root crontab removed"

# 5. Confirm Dashboard is still healthy after tmux kill
```
