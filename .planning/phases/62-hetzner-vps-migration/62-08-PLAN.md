backup Dockerfile (from RESEARCH.md, adapted):
```dockerfile
FROM ubuntu:22.04
ENV DEBIAN_FRONTEND=noninteractive
RUN apt-get update && apt-get install -y \
  postgresql-client-16 \
  python3 \
  python3-pip \
  cron \
  && pip3 install b2 \
  && apt-get clean \
  && rm -rf /var/lib/apt/lists/*
COPY backup.sh /backup.sh
RUN chmod +x /backup.sh
# Set up cron and run in foreground
ENTRYPOINT ["/bin/bash", "-c", "echo '0 2 * * * /backup.sh >> /var/log/backup.log 2>&1' | crontab - && service cron start && tail -f /var/log/backup.log"]
```

backup.sh (from RESEARCH.md Pattern 5):
```bash
#!/bin/bash
set -eux
BACKUP_DIR="/backups"
TIMESTAMP=$(date -u +%Y%m%d-%H%M%S)
mkdir -p "$BACKUP_DIR"

for db in reforma ynab kidai; do
  echo "[$(date -u +%FT%TZ)] Dumping $db..."
  PGPASSWORD="$POSTGRES_PASSWORD" pg_dump \
    -h "${db}-db" \
    -U postgres \
    "$db" | gzip > "$BACKUP_DIR/${db}-${TIMESTAMP}.sql.gz"
  echo "[$(date -u +%FT%TZ)] Dump complete: ${db}-${TIMESTAMP}.sql.gz ($(du -sh $BACKUP_DIR/${db}-${TIMESTAMP}.sql.gz | cut -f1))"
done

# Also back up GSD Dashboard SQLite
echo "[$(date -u +%FT%TZ)] Backing up dashboard.db..."
cp /dashboard-data/dashboard.db "$BACKUP_DIR/dashboard-${TIMESTAMP}.db" 2>/dev/null || \
  echo "Warning: dashboard.db not found — SQLite backup skipped"

echo "[$(date -u +%FT%TZ)] Uploading to Backblaze B2..."
b2 authorize-account "$B2_APPLICATION_KEY_ID" "$B2_APPLICATION_KEY"
b2 sync --keepDays "$RETENTION_DAYS" "$BACKUP_DIR/" "b2://${B2_BUCKET_NAME}/backups/"

echo "[$(date -u +%FT%TZ)] Cleaning up local backups..."
rm -f "$BACKUP_DIR"/*.sql.gz "$BACKUP_DIR"/*.db

echo "[$(date -u +%FT%TZ)] Backup complete."
```

Docker Compose backup service entry:
```yaml
  backup:
    build:
      context: /home/services/backups
      dockerfile: Dockerfile
    container_name: backup
    environment:
      - B2_APPLICATION_KEY_ID=${B2_APPLICATION_KEY_ID}
      - B2_APPLICATION_KEY=${B2_APPLICATION_KEY}
      - B2_BUCKET_NAME=${B2_BUCKET_NAME}
      - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
      - RETENTION_DAYS=30
    volumes:
      - /var/tmp/backups:/backups
      - dashboard-data:/dashboard-data:ro
    depends_on:
      - reforma-db
      - ynab-db
      - kidai-db
    restart: unless-stopped
    logging:
      driver: "json-file"
      options:
        max-size: "10m"
        max-file: "3"
```

  postgresql-client-16 \

  cron \

  && apt-get clean \
  && rm -rf /var/lib/apt/lists/*

ENTRYPOINT ["/bin/bash", "-c", "echo '0 2 * * * /backup.sh >> /var/log/backup.log 2>&1' | crontab - && service cron start && tail -f /var/log/backup.log"]
DOCKEREOF

# 2. Write backup.sh
cat > /home/services/backups/backup.sh << 'SCRIPTEOF'
#!/bin/bash
set -eux
BACKUP_DIR="/backups"
TIMESTAMP=$(date -u +%Y%m%d-%H%M%S)
mkdir -p "$BACKUP_DIR"

for db in reforma ynab kidai; do
  echo "[$(date -u +%FT%TZ)] Dumping $db..."
  PGPASSWORD="$POSTGRES_PASSWORD" pg_dump \
    -h "${db}-db" \
    -U postgres \
    "$db" | gzip > "$BACKUP_DIR/${db}-${TIMESTAMP}.sql.gz"
  echo "[$(date -u +%FT%TZ)] Dump: ${db}-${TIMESTAMP}.sql.gz"
done

echo "[$(date -u +%FT%TZ)] Backing up dashboard SQLite..."
cp /dashboard-data/dashboard.db "$BACKUP_DIR/dashboard-${TIMESTAMP}.db" 2>/dev/null \
  || echo "Warning: dashboard.db not found at /dashboard-data — SQLite backup skipped"

echo "[$(date -u +%FT%TZ)] Uploading to B2..."
b2 authorize-account "$B2_APPLICATION_KEY_ID" "$B2_APPLICATION_KEY"
b2 sync --keepDays "$RETENTION_DAYS" "$BACKUP_DIR/" "b2://${B2_BUCKET_NAME}/backups/"

echo "[$(date -u +%FT%TZ)] Cleanup..."
rm -f "$BACKUP_DIR"/*.sql.gz "$BACKUP_DIR"/*.db 2>/dev/null || true

echo "[$(date -u +%FT%TZ)] Backup complete."
SCRIPTEOF

chmod +x /home/services/backups/backup.sh

echo "Backup files created:"
ls -la /home/services/backups/
BACKUPEOF

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| backup container → DB containers | Internal Docker bridge network; reads via PGPASSWORD env var |
| backup container → Backblaze B2 | HTTPS; authenticated via B2_APPLICATION_KEY_ID + B2_APPLICATION_KEY |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-62-27 | Disclosure | B2_APPLICATION_KEY in container env | mitigate | Sourced from .env.production (not in image); masked in `docker inspect` via `--env-file`; key scoped to gsdlabs-backups bucket only (minimal permissions) |
| T-62-28 | Disclosure | Backup dumps stored temporarily in /var/tmp/backups | mitigate | Deleted immediately after B2 upload in backup.sh; /var/tmp is not a persistent volume |
| T-62-29 | Denial of Service | Disk fills with dumps if B2 upload fails | mitigate | Script uses `set -eux` — exits on any failure; cleanup runs only after successful upload; /var/tmp has limited space; RESEARCH.md pitfall 5 addressed |
| T-62-30 | Disclosure | B2 bucket is public | mitigate | Backblaze B2 bucket must be Private (no public access); confirmed in user_setup of Plan 01 |
</threat_model>
