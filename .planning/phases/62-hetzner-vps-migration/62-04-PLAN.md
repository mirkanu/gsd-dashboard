```bash
# 1. Clone GSD Dashboard repo
cd /home/services
git clone https://github.com/manuelkuhs/gsd-dashboard.git gsddashboard
cd gsddashboard

# 2. Install production dependencies
npm ci --production

# 3. Build if needed (check if build step exists)
npm run build 2>/dev/null || echo "No build step required"

# 4. Create .env file on VPS from the master .env.production
# Copy relevant vars from /home/services/.env.production

NODE_ENV=production
DASHBOARD_PORT=4820
TELEGRAM_CHAT_ID={value from /home/services/.env.production TELEGRAM_CHAT_ID}
CLOUDFLARE_API_KEY={value from /home/services/.env.production CLOUDFLARE_API_KEY}
CLOUDFLARE_EMAIL={value from /home/services/.env.production CLOUDFLARE_EMAIL}
# GSD_DATA_URL is now fixed (no quick tunnel rotation needed)
GSD_DATA_URL=https://dashboard.gsdlabs.dev
ENVEOF
chmod 600 /home/services/gsddashboard/.env

# 5. Create ecosystem config for PM2
cat > /home/services/gsddashboard/ecosystem.config.cjs << 'ECOSYSTEMEOF'
module.exports = {
  apps: [
    {
      name: 'gsd-dashboard',
      script: 'server/index.js',
      cwd: '/home/services/gsddashboard',
      interpreter: 'node',
      env: {
        NODE_ENV: 'production',
        DASHBOARD_PORT: '4820',
      },
      restart_delay: 3000,
      max_restarts: 10,
      log_date_format: 'YYYY-MM-DD HH:mm:ss',
    },
    {
      name: 'gsd-healthcheck',
      script: 'scripts/healthcheck.sh',
      cwd: '/home/services/gsddashboard',
      interpreter: 'bash',
      restart_delay: 5000,
    }
  ]
};
ECOSYSTEMEOF

# 6. Create data directory for SQLite
mkdir -p /home/services/gsddashboard/data

# 7. Start dashboard and healthcheck via PM2
pm2 start /home/services/gsddashboard/ecosystem.config.cjs
pm2 save

# 8. Verify dashboard is running
sleep 5
curl -f http://localhost:4820/api/health
echo "Dashboard health check passed"

pm2 list
```
