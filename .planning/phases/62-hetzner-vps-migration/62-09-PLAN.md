```bash
VPS_IP="{VPS_IP from Plan 01}"

echo "=== VPS Service Health Check ==="
echo "Run at: $(date -u)"

# 1. Docker Compose status
echo "--- docker compose ps ---"
ssh root@$VPS_IP "cd /home/services/hetzner-vps && docker compose --env-file /home/services/.env.production ps"

# 2. PM2 status
echo "--- pm2 list ---"
ssh root@$VPS_IP "pm2 list"

# 3. Health checks per service (via gsdlabs.dev subdomains — external)
echo "--- External subdomain checks ---"
for url in \
  "https://dashboard.gsdlabs.dev/api/health" \
  "https://debates.gsdlabs.dev/" \
  "https://ynab.gsdlabs.dev/api/webhook" \
  "https://kidai.gsdlabs.dev/" \
  "https://api.gsdlabs.dev/" \
  "https://reforma-coral.vercel.app/"; do
  STATUS=$(curl -s -o /dev/null -w "%{http_code}" --max-time 10 "$url" 2>/dev/null || echo "TIMEOUT")
  echo "  $url → $STATUS"
done

# 4. Local port checks (via SSH)
echo "--- Local port checks on VPS ---"
for port_service in "4820:gsd-dashboard" "3000:debates" "3001:ynab-api" "3002:kidai-admin" "9000:image-search-mcp"; do
  PORT="${port_service%%:*}"
  SVC="${port_service##*:}"
  STATUS=$(ssh root@$VPS_IP "curl -s -o /dev/null -w '%{http_code}' --max-time 5 http://localhost:$PORT/ 2>/dev/null || echo 'FAIL'")
  echo "  localhost:$PORT ($SVC) → $STATUS"
done

# 5. Database connectivity
echo "--- Database pg_isready ---"
for db in reforma-db ynab-db kidai-db; do
  RESULT=$(ssh root@$VPS_IP "docker exec $db pg_isready -U postgres 2>&1")
  echo "  $db: $RESULT"
done

# 6. B2 backup files
echo "--- B2 backup files ---"
ssh root@$VPS_IP "docker exec backup b2 ls b2://{B2_BUCKET_NAME}/backups/ 2>/dev/null | tail -10" || \
  echo "  (manual check required: log into B2 console)"

# 7. KidAI crontab
echo "--- KidAI crontab on VPS ---"
ssh root@$VPS_IP "crontab -l | grep kidai"

# 8. GitHub Actions deploy status (last run per repo)
echo "--- GitHub Actions deploy statuses ---"
for repo in gsddashboard debates ynab KidAI; do
  cd /data/home/$repo 2>/dev/null || continue
  STATUS=$(gh run list --limit=1 --json conclusion --jq '.[0].conclusion' 2>/dev/null || echo "unknown")
  echo "  $repo: $STATUS"
done
```
