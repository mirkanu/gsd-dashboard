Step 1 — Create the GitHub Actions directory and deploy workflow file for debates.
Note: debates may not have a .github/workflows/ directory. Create it:

```bash
mkdir -p /data/home/debates/.github/workflows
```

Write the deploy workflow:
```yaml
# /data/home/debates/.github/workflows/deploy-hetzner.yml
name: Deploy to Hetzner VPS

on:
  push:
    branches: [main]
  workflow_dispatch:

env:
  SERVICE_NAME: debates
  SERVICE_PATH: /home/services/debates

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Deploy via SSH to Hetzner VPS
        uses: appleboy/ssh-action@v1.0.0
        with:
          host: ${{ secrets.HETZNER_VPS_IP }}
          username: root
          key: ${{ secrets.HETZNER_SSH_KEY }}
          port: 22
          script: |
            set -eux

            cd ${{ env.SERVICE_PATH }}
            git fetch origin main
            git checkout origin/main

            cd /home/services/hetzner-vps
            docker compose --env-file /home/services/.env.production \
              up --build -d ${{ env.SERVICE_NAME }}

            sleep 10
            curl -f http://localhost:3000/ || exit 1
            echo "Debates deployment successful"
```
