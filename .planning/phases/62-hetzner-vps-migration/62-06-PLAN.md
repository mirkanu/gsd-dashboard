Step 2 — Dump Railway ynab database from this local machine.
Use the Railway ynab DATABASE_URL collected in Plan 01 checkpoint:

```bash
# Replace {YNAB_RAILWAY_DATABASE_URL} with the Railway URL from Plan 01 checkpoint
YNAB_RAILWAY_DB="{YNAB_RAILWAY_DATABASE_URL}"
pg_dump "$YNAB_RAILWAY_DB" -Fc -f /tmp/ynab-railway-dump.dump
ls -lh /tmp/ynab-railway-dump.dump
```

Step 3 — Copy dump to VPS and restore into ynab-db:
```bash
scp /tmp/ynab-railway-dump.dump root@{VPS_IP}:/tmp/ynab-railway-dump.dump

ssh root@{VPS_IP} << 'RESTORE_EOF'
set -eux
# Copy dump file into container and restore
docker cp /tmp/ynab-railway-dump.dump ynab-db:/tmp/ynab-dump.dump
docker exec ynab-db pg_restore \
  -U postgres \
  -d ynab \
  --no-owner \
  --no-acl \
  --exit-on-error \
  /tmp/ynab-dump.dump

# Verify row count in a key table
docker exec ynab-db psql -U postgres -d ynab -c "\dt" 2>/dev/null
docker exec ynab-db psql -U postgres -d ynab -c "SELECT COUNT(*) FROM (SELECT tablename FROM pg_tables WHERE schemaname='public') t;"

# Clean up
rm -f /tmp/ynab-railway-dump.dump
docker exec ynab-db rm -f /tmp/ynab-dump.dump
RESTORE_EOF

# Clean up local dump
rm -f /tmp/ynab-railway-dump.dump
```

Step 1 — Create the deploy workflow for ynab:

```bash
mkdir -p /data/home/ynab/.github/workflows
```

Write /data/home/ynab/.github/workflows/deploy-hetzner.yml:
```yaml
name: Deploy to Hetzner VPS

on:
  push:
    branches: [main]
  workflow_dispatch:

env:
  SERVICE_NAME: ynab-api
  SERVICE_PATH: /home/services/ynab

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Deploy via SSH to Hetzner VPS
        uses: appleboy/ssh-action@v1.0.0
        with:
          host: ${{ secrets.HETZNER_VPS_IP }}
          username: root
          key: ${{ secrets.HETZNER_SSH_KEY }}
          port: 22
          script: |
            set -eux

            cd ${{ env.SERVICE_PATH }}
            git fetch origin main
            git checkout origin/main

            cd /home/services/hetzner-vps
            docker compose --env-file /home/services/.env.production \
              up --build -d ${{ env.SERVICE_NAME }}

            sleep 15
            curl -f http://localhost:3001/api/webhook || exit 1
            echo "ynab deployment successful"
```
