  </behavior>

Create `server/crypto.js`. Use `node:crypto`: `createHash('sha256')` for key derivation, `randomBytes(12)` for IV, `createCipheriv('aes-256-gcm', key, iv)` and `createDecipheriv` with `setAuthTag`. Store ciphertext/IV/tag as base64 strings (not hex — smaller on disk).

Pattern:
```javascript
const crypto = require('node:crypto');
const { db } = require('./db');

function getKey() {
  const raw = process.env.DASHBOARD_SECRET_KEY;
  if (!raw) throw new Error('DASHBOARD_SECRET_KEY not set — cannot encrypt credentials');
  return crypto.createHash('sha256').update(raw).digest(); // 32 bytes
}

function encryptField(plaintext) { /* ... */ }
function decryptField({ value_encrypted, iv, auth_tag }) { /* ... */ }

const upsertSecret = db.prepare(`
  INSERT INTO app_settings (key, value_encrypted, iv, auth_tag, is_secret, updated_at)
  VALUES (?, ?, ?, ?, 1, strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
  ON CONFLICT(key) DO UPDATE SET
    value_encrypted = excluded.value_encrypted,
    iv = excluded.iv,
    auth_tag = excluded.auth_tag,
    updated_at = excluded.updated_at
`);
const selectSecret = db.prepare('SELECT * FROM app_settings WHERE key = ?');
const listSecretKeysStmt = db.prepare('SELECT key, updated_at FROM app_settings WHERE is_secret = 1');

function setSecret(key, plaintext) { const { value_encrypted, iv, auth_tag } = encryptField(plaintext); upsertSecret.run(key, value_encrypted, iv, auth_tag); }
function getSecret(key) { const row = selectSecret.get(key); if (!row) return null; try { return decryptField(row); } catch (e) { console.error('[crypto] decrypt failed for', key, e.message); return null; } }
function listSecretKeys() { return listSecretKeysStmt.all(); }

module.exports = { encryptField, decryptField, getSecret, setSecret, listSecretKeys };
```
