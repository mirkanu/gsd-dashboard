// GET /api/services/costs?month=YYYY-MM
interface CostsResponse {
  month: string; // "2026-04"
  services: Array<{ service: string; total_usd: number; source_breakdown: Record<string, number> }>;
  projects: Array<{ project_key: string | null; total_usd: number; services: Array<{ service: string; total_usd: number }> }>;
  needs_review: Array<{ id: string; service: string; cost_usd: number; raw_body: string | null; checked_at: string }>;
  entries: Array<{
    id: string; service: string; cost_usd: number; currency: string;
    project_key: string | null; source: 'manual'|'email'|'recurring'|'unparsed'|'api';
    description: string; checked_at: string; message_id: string | null; notes: string | null;
  }>;
}

// POST /api/services/costs
interface CreateCostBody {
  service: string;
  project_key?: string | null;
  cost_usd: number;
  currency?: string;
  start_date?: string;
  recurring_monthly?: boolean;
  notes?: string;
  description?: string;
}

// DELETE /api/services/costs/:id — operates on external_service_costs.id uniformly.
// Handles source IN ('manual','unparsed','recurring','email'). Server cascades via notes convention.

// GET /api/services/rules → { rules: Array<Rule> }
interface Rule { id: number; pattern_type: 'sender'|'subject_contains'; pattern_value: string; project_key: string; service: string | null; created_at: string }

// GET /api/app-settings → { keys: Array<{ key: string; updated_at: string; set: true }> }
// PUT /api/app-settings/:key body { value: string } → { ok: true }
// DELETE /api/app-settings/:key → { ok: true }

services: {

  costs: {
    get: (month?: string) => request<CostsResponse>(`/services/costs${month ? `?month=${month}` : ''}`),
    create: (body: CreateCostBody) => request<{ok:true, id:string|null, manual_id:string}>(`/services/costs`, { method: 'POST', body: JSON.stringify(body) }),
    update: (id: string, body: Partial<CreateCostBody> & { source?: string }) => request<{ok:true}>(`/services/costs/${id}`, { method: 'PATCH', body: JSON.stringify(body) }),
    delete: (id: string) => request<{ok:true, deleted:number}>(`/services/costs/${id}`, { method: 'DELETE' }),
  },
  rules: {
    list: () => request<{rules:Rule[]}>(`/services/rules`),
    create: (body: Omit<Rule,'id'|'created_at'>) => request<{id:number}>(`/services/rules`, { method: 'POST', body: JSON.stringify(body) }),
    update: (id: number, body: Partial<Rule>) => request<{ok:true}>(`/services/rules/${id}`, { method: 'PATCH', body: JSON.stringify(body) }),
    delete: (id: number) => request<{ok:true}>(`/services/rules/${id}`, { method: 'DELETE' }),
  },

appSettings: {
