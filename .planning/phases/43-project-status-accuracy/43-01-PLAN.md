<objective>
Fix the false "Waiting" bug in tmux state detection and add a clean current-task extractor.

Purpose: `detectSessionStateAsync` currently only returns 'working' when specific Claude Code timer UI strings (`· ↓`, `(thinking)`) are visible. When Claude is producing output via tool use, file edits, or other activity without those exact strings, it falls through to 'waiting' — this is the core STAT-02 bug the user sees daily on the live Railway URL. This plan fixes it with a secondary "output is changing" heuristic and expands pattern coverage. It also adds `extractCurrentTask` to produce a meaningful task preview line for STAT-04.

Output: Updated `server/gsd/tmux.js` with new detection logic + new export `extractCurrentTask`, plus unit tests proving the fixes.
</objective>

// Output-change heuristic: if capture-pane content differs from last capture
// within 3 seconds, session is actively producing output even when timer UI
// isn't visible (tool use, file edits, etc.). Fixes false "Waiting" bug.
const crypto = require('crypto');
const hash = crypto.createHash('sha1').update(output).digest('hex').slice(0, 16);
const prev = paneHashCache.get(sessionName);
const nowMs = Date.now();
if (prev && prev.hash !== hash) {
  // Content changed since last call — update and declare working
  paneHashCache.set(sessionName, { hash, lastChangedAt: nowMs });
  return 'working';
}
if (prev && prev.hash === hash && (nowMs - prev.lastChangedAt) < 3000) {
  // Same content but recent change within window — still working
  return 'working';
}
// Initialize or stale — fall through to waiting detection
if (!prev) paneHashCache.set(sessionName, { hash, lastChangedAt: nowMs });
```

/**
 * Extract a meaningful "current task" preview line from a tmux capture-pane buffer.
 * Scans from bottom up in the last 20 lines, skips UI chrome (prompts, shortcuts,
 * box drawing, numeric selection menus, known Claude Code footer), strips leading
 * marker characters, and returns the first meaningful line (>= 4 chars after cleanup).
 * Returns null if no such line is found. Max 120 chars; longer lines truncated with "…".
 * @param {string|null|undefined} rawText
 * @returns {string|null}
 */
function extractCurrentTask(rawText) {
  if (!rawText) return null;
  const lines = rawText.split('\n').slice(-20);
  const chromePatterns = [
    /esc\s+to\s+interrupt/i,
    /\?\s+for\s+shortcuts/i,
    /^\s*\(y\/n\)/i,
    /Bypassing\s+Permissions/i,
    /^[-─═\s]+$/,
    /^\s*>\s*\d+\.\s/,                 // numeric menu "> 1. Option"
    /^[✻✶]/,                            // status spinner lines (extractStatusLine handles those)
  ];
  for (let i = lines.length - 1; i >= 0; i--) {
    let clean = stripAnsi(lines[i]).trim();
    if (!clean) continue;
    if (chromePatterns.some(p => p.test(clean))) continue;
    // Strip leading marker characters
    clean = clean.replace(/^[│|>·⏺\s]+/, '').trim();
    if (clean.length < 4) continue;
    if (clean.length > 120) clean = clean.slice(0, 119) + '…';
    return clean;
  }
  return null;
}
```
