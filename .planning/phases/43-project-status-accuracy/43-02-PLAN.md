<objective>
Add a server-side background state broadcaster that pushes real-time project state changes over the existing WebSocket and augments the projects API with `stateEnteredAt` and `currentTask` fields.

Purpose: Today the client polls `/api/gsd/projects` every 10-60s (GSD.tsx line 955-961) and only sees state changes on the next tick. STAT-01 requires sub-second visibility of state transitions — the only way to achieve that without hammering the API is to push from server. This plan creates the push source + the data the client needs for STAT-03 (elapsed time) and STAT-04 (task preview) on the HTTP fallback path.

Output:
- New `server/gsd/stateBroadcaster.js` module that polls tmux sessions every 2s, tracks state transitions + `stateEnteredAt` timestamps, extracts current task, and broadcasts on change.
- `server/routes/gsd.js` projects response augmented with `stateEnteredAt` and `currentTask` (merged from snapshot).
- `server/index.js` starts the broadcaster at boot (only when running as data source, not as Railway proxy).
</objective>

```js
'use strict';

const { detectSessionStateAsync, capturePaneTextAsync, extractCurrentTask, extractStatusLine } = require('./tmux');

// In-memory snapshot: projectName -> { sessionState, stateEnteredAt, currentTask, statusText }
const snapshot = new Map();

const DEFAULT_POLL_INTERVAL_MS = 2000;

/**
 * Poll a single project once and, if state changed, update snapshot + broadcast.
 * Exposed as _testPollOnce for unit tests via dependency injection.
 * @param {{ name: string, tmux_session: string|null, archived?: boolean }} project
 * @param {Function} detectFn - async (sessionName) => state
 * @param {Function} captureFn - async (sessionName) => string|null
 * @param {Function} broadcastFn - (type, data) => void
 * @returns {Promise<boolean>} true if a transition was broadcast
 */
async function _testPollOnce(project, detectFn, captureFn, broadcastFn) {
  if (!project || project.archived || !project.tmux_session) return false;
  let sessionState;
  let paneText = null;
  try {
    sessionState = await detectFn(project.tmux_session);
    paneText = await captureFn(project.tmux_session);
  } catch {
    return false; // transient — skip tick, preserve snapshot
  }
  const currentTask = extractCurrentTask(paneText);
  const statusText = sessionState === 'working' ? (extractStatusLine(paneText) || null) : null;

  const prev = snapshot.get(project.name);
  const nowIso = new Date().toISOString();

  if (!prev) {
    // Initial seed — record silently, no broadcast
    snapshot.set(project.name, { sessionState, stateEnteredAt: nowIso, currentTask, statusText });
    return false;
  }
  if (prev.sessionState === sessionState) {
    // Same state — keep stateEnteredAt, refresh currentTask / statusText
    snapshot.set(project.name, { sessionState, stateEnteredAt: prev.stateEnteredAt, currentTask, statusText });
    return false;
  }
  // Transition — broadcast + update
  const entry = { sessionState, stateEnteredAt: nowIso, currentTask, statusText };
  snapshot.set(project.name, entry);
  broadcastFn('project_state_change', {
    project: project.name,
    sessionState,
    statusText,
    currentTask,
    stateEnteredAt: nowIso,
  });
  return true;
}

/**
 * Start the background poller. Fire-and-forget — never blocks startup.
 * @param {Function} loadProjectsFn - () => { projects: Array }
 * @param {Function} broadcastFn - (type, data) => void
 * @param {number} [intervalMs]
 * @returns {Function} stop function (for tests)
 */
function startStateBroadcaster(loadProjectsFn, broadcastFn, intervalMs = DEFAULT_POLL_INTERVAL_MS) {
  let stopped = false;
  async function tick() {
    if (stopped) return;
    try {
      const { projects } = loadProjectsFn();
      await Promise.all(projects.map(p =>
        _testPollOnce(p, detectSessionStateAsync, capturePaneTextAsync, broadcastFn)
      ));
    } catch { /* config read failure — retry next tick */ }
    if (!stopped) setTimeout(tick, intervalMs);
  }
  // Kick off first tick after a small delay so server startup is not blocked
  setTimeout(tick, 500);
  return () => { stopped = true; };
}

function getProjectStateSnapshot() {
  const out = {};
  for (const [name, entry] of snapshot.entries()) out[name] = { ...entry };
  return out;
}

function _resetSnapshot() { snapshot.clear(); }

module.exports = {
  startStateBroadcaster,
  getProjectStateSnapshot,
  _testPollOnce,
  _resetSnapshot,
};
```
