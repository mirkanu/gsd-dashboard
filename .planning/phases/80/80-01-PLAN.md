```javascript
const path = require('path');

const PLUGIN_SCRIPTS = '/home/claude/.claude/plugins/marketplaces/thedotmack/claude-mem/plugin/scripts';

module.exports = {
  apps: [{
    name: 'claude-mem-worker',
    script: path.join(PLUGIN_SCRIPTS, 'worker-service.cjs'),
    args: 'start',
    interpreter: '/home/claude/.bun/bin/bun',

    max_memory_restart: '512M',
    restart_delay: 3000,
    max_restarts: 10,
    min_uptime: '10s',
  }],
};
```

```bash
cd /home/services && pm2 start /home/services/.ecosystem-mem.cjs
```
