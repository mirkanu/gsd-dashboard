    Create server/routes/autopilot.js:

    ```javascript
    'use strict';
    const express = require('express');
    const path = require('path');
    const fs = require('fs');
    const { AutopilotManager } = require('../autopilot/AutopilotManager');

    const router = express.Router();

    // In-memory registry: projectName → { manager: AutopilotManager, runId: string }
    // One active run per project at a time.
    const runRegistry = new Map();

    function loadConfig() {
      const configPath = process.env.GSD_PROJECTS_PATH ||
        path.join(__dirname, '../../gsd-projects.json');
      return JSON.parse(fs.readFileSync(configPath, 'utf8'));
    }

    function findProject(projectName) {
      const config = loadConfig();
      return config.projects.find(p => p.name === projectName) || null;
    }
    ```

    POST /api/autopilot/start:
    - Validate projectName in body, 400 if missing
    - If runRegistry has an active run for this project, return 409 { error: 'Run already active' }
    - new AutopilotManager(), await manager.start(projectName, { runType: mode || 'execute' })
    - Store { manager, runId } in runRegistry under projectName
    - Return 200 { runId, status: 'running' }

    POST /api/autopilot/pause:
    - Look up runRegistry by projectName, 404 if not found
    - manager.pause()
    - Return 200 { ok: true }

    POST /api/autopilot/resume:
    - Look up runRegistry, 404 if not found
    - manager.resume()
    - Return 200 { ok: true }

    GET /api/autopilot/status/:projectName:
    - Look up runRegistry; if not found return { runId: null, status: 'idle', projectName }
    - Otherwise return manager.getStatus()

    POST /api/autopilot/plan-all:
    - Same as /start but hardcodes mode = 'plan-all'
    - Plan-all mode: AutopilotManager runs only /gsd:plan-phase loops without /gsd:execute-phase
