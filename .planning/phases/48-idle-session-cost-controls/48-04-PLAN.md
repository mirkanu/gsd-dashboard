    In the existing useEffect (or a separate one), fetch project costs:
    ```tsx
    async function fetchIdleCosts() {
      try {
        const gsdData = await api.get('/api/gsd/projects');
        // Filter to waiting sessions (from project state, if available in the response)
        // Or fetch all costs and show total (simpler — don't need to filter by state)
        const costResults = await Promise.allSettled(
          (gsdData.projects || []).map(async (p: any) => {
            if (p.tmux_session) {
              return api.get(`/api/gsd/projects/${encodeURIComponent(p.name)}/tmux-cost`);
            }
            return null;
          })
        );
        let total = 0, count = 0;
        for (const r of costResults) {
          if (r.status === 'fulfilled' && r.value?.dailyCostUsd > 0) {
            total += r.value.dailyCostUsd;
            count++;
          }
        }
        setIdleCostPerDay(total);
        setIdleSessionCount(count);
      } catch {
        // Silent — banner is non-critical
      }
    }
    ```

          {idleSessionCount} session{idleSessionCount !== 1 ? 's' : ''} running
