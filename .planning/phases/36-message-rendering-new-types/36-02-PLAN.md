**3. Terminal content detection heuristic:**

Before rendering markdown, check if content looks like terminal output using a simple heuristic:
```typescript
function looksLikeTerminal(text: string): boolean {
  const lines = text.split('\n');
  if (lines.length < 2) return false;
  // ASCII table: multiple lines with | or + alignment chars
  const pipeLines = lines.filter(l => /\|.*\|/.test(l) && /[-+|]/.test(l)).length;
  if (pipeLines >= 3) return false; // Let markdown handle real tables
  // Box-drawing characters (─ │ ┌ ┐ └ ┘ ├ ┤)
  const boxChars = lines.filter(l => /[─│┌┐└┘├┤┬┴┼═║╔╗╚╝╠╣╦╩╬]/.test(l)).length;
  if (boxChars >= 2) return true;
  // Indentation-heavy: >60% of lines start with 2+ spaces
  const indented = lines.filter(l => /^ {2,}\S/.test(l)).length;
  if (lines.length >= 3 && indented / lines.length > 0.6) return true;
  return false;
}
```

If `looksLikeTerminal(msg.content)` returns true, render in a monospace pre block instead of markdown:

