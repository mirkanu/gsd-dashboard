```typescript
const handleTouchMove = (e: TouchEvent) => {
  const dy = touchStartY - e.touches[0].clientY;
  const dx = touchStartX - e.touches[0].clientX;
  if (!scrollIntent) {
    if (Math.abs(dy) < 5) return;
    scrollIntent = Math.abs(dy) >= Math.abs(dx);
  }
  if (!scrollIntent) return;
  e.preventDefault();
  e.stopImmediatePropagation();
  touchStartY = e.touches[0].clientY;
  const lines = Math.max(1, Math.abs(Math.round(dy / ((terminal.options.fontSize as number) ?? 14))));
  const seq = dy > 0 ? '\x1b[<64;1;1M' : '\x1b[<65;1;1M';
  for (let i = 0; i < lines; i++) {
    if (ws.readyState === WebSocket.OPEN) ws.send(seq);
  }
};
```


To:
```html
<meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1" />
```

Note: `maximum-scale=1` (without `user-scalable=no`) is the correct fix. It stops auto-zoom on focus while still allowing pinch-to-zoom for accessibility. `user-scalable=no` would block intentional zoom — do NOT use it.

This fixes MOB-02: the iOS WebKit engine auto-zooms whenever a focused input has font-size less than 16px. The xterm.js helper textarea is 16px but some other inputs in the overlay may be smaller. Adding maximum-scale=1 to the global viewport meta is the canonical fix.

**Fix 1 — MOB-01: Touch scroll damping**

Add a `SCROLL_DAMPING` constant near the top of the `useEffect` inside `TerminalOverlay` (or just above `handleTouchMove`):

```typescript
const SCROLL_DAMPING = 3; // pixels of drag per tmux scroll line (higher = slower/more deliberate)
```

Then in `handleTouchMove`, replace the `lines` calculation:
```typescript
// BEFORE:
const lines = Math.max(1, Math.abs(Math.round(dy / ((terminal.options.fontSize as number) ?? 14))));

// AFTER:
const fontSize = (terminal.options.fontSize as number) ?? 10;
const lines = Math.max(1, Math.abs(Math.round(dy / (fontSize * SCROLL_DAMPING))));
```

Explanation: The original `dy / fontSize` at mobile fontSize=10 means just 10px of finger movement sends 1 scroll event per loop iteration, and the loop runs on every touchmove. Multiplying the denominator by `SCROLL_DAMPING=3` means 30px of drag per line. This produces a comfortable scroll speed. Adjust the constant higher to slow down further if needed.

**Fix 2 — MOB-03: Re-focus terminal after special key tap**

Update the `SpecialKeyBar` component to accept a `termRef` prop and call `terminal.focus()` after sending the key sequence. This ensures the xterm.js textarea regains focus after each special key tap, preventing scroll position from shifting due to focus jumping to another element.

Change the `SpecialKeyBar` function signature:
```typescript
// BEFORE:
function SpecialKeyBar({ wsRef }: { wsRef: React.RefObject<WebSocket | null> }) {

// AFTER:
function SpecialKeyBar({ wsRef, termRef }: { wsRef: React.RefObject<WebSocket | null>; termRef: React.RefObject<Terminal | null> }) {
```

Change the `send` function inside `SpecialKeyBar`:
```typescript
// BEFORE:
const send = (seq: string) => {
  if (wsRef.current?.readyState === WebSocket.OPEN) {
    wsRef.current.send(seq);
  }
};

// AFTER:
const send = (seq: string) => {
  if (wsRef.current?.readyState === WebSocket.OPEN) {
    wsRef.current.send(seq);
  }
  // Re-focus terminal so scroll position and input focus stay on the terminal,
  // not on the button that was just tapped.
  termRef.current?.focus();
};
```

Update the usage site in `TerminalOverlay` where `SpecialKeyBar` is rendered:

The `termRef.current?.focus()` call is an explicit re-focus to guard against any edge cases where xterm.js loses focus anyway.
