    Step 1 — Check available memory before launching Playwright:
    `free -m | awk '/Mem:/ {print $7}'`
    If the value is below 800, skip Playwright and note "SKIPPED: insufficient memory (Xmb free)".

    Step 2 — If memory is sufficient, write the following script to /tmp/playwright-54b-uat.js
    and run it with: `NODE_PATH=/usr/lib/node_modules node /tmp/playwright-54b-uat.js`

    Script contents (write with Write tool, not heredoc):
    ```js
    const http = require('http');

    const body = JSON.stringify({
      script: `
        const dashPass = process.env.DASHBOARD_PASS || '';
        await page.goto('https://dashboard.gsdlabs.dev/login');
        const passInput = page.locator('input[type="password"]');
        if (await passInput.count() > 0) {
          await passInput.fill(dashPass);
          await page.locator('button[type="submit"]').click();
          await page.waitForNavigation({ timeout: 10000 }).catch(() => {});
        }
        await page.goto('https://dashboard.gsdlabs.dev/config');
        await page.waitForSelector('body', { timeout: 10000 });
        const html = await page.content();
        const hasPanel = html.includes('Notification Policy') || html.includes('NotificationPolicy');
        const hasCard = html.includes('notification') && html.includes('policy');
        return { url: page.url(), hasPanel, hasCard, titleFound: hasPanel };
      `
    });

    const opts = {
      hostname: 'localhost',
      port: 3099,
      path: '/job',
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(body) }
    };

    const req = http.request(opts, res => {
      let data = '';
      res.on('data', c => data += c);
      res.on('end', () => {
        const r = JSON.parse(data);
        if (r.error) { console.error('Playwright error:', r.error); process.exit(1); }
        console.log('UAT result:', JSON.stringify(r.result, null, 2));
        if (r.result && r.result.hasPanel) {
          console.log('PASS: NotificationPolicyPanel visible at /config');
        } else {
          console.log('FAIL: NotificationPolicyPanel NOT found — check bundle');
        }
      });
    });
    req.on('error', e => { console.error('Daemon error:', e.message); process.exit(1); });
    req.write(body);
    req.end();
    ```

    Note: DASHBOARD_PASS is available in /home/services/.env.production — source it before

    `. /home/services/.env.production && NODE_PATH=/usr/lib/node_modules node /tmp/playwright-54b-uat.js`
